## Method

- CPU attribution: `cpusample.py` polls `ps -Ao pid=,ppid=,time=,command=` every 2 s
  for 300 s and sums cumulative CPU-time deltas per pid (pids born inside the window
  count their whole CPU time). Processes that live under 2 s are invisible to it.
- Spawn churn: `spawnsample.py` polls `ps` at 0.5 s and counts newly seen pids by
  command and parent; `pid_delta` (max-pid growth) bounds all spawns.
- Containers: Docker Engine API `/containers/<name>/stats` (same data as
  `docker stats`), two reads 120 s apart, for CPU-ns and network/block byte deltas.
- Hub clients: `lsof -nP -iTCP:60891 -sTCP:ESTABLISHED` on the host (socket table
  only; no DB connection).
- LLM spawns: `~/.gobby/logs/llm.log` `feature_llm_call` lines grouped by feature.

The scripts are inlined in Appendix A, and the after-measurement commands are in
"Live after-measurement at cutover".

| --- | ---: | ---: | ---: |
| pre-tool-use | 24 | 6 (registration cache, transcript recheck, activation reconcile, templating project info, turn lifecycle, event enrichment) | 3 (state_manager get x2, contested expiry) |
| post-tool-use | 27 | 6 (same family plus observer_context_usage) | 4 (get, merge, resolve_open_tool_errors, contested expiry) |

Every `sessions` read is `SELECT *` of the full row (summary_markdown,

interactive sessions every 5 s.

Live blob size sample: the Orchestrator seat's `_observations` list alone holds about
300 `track-task-claim` entries with empty messages (about 33 KB) and grows without
bound; the whole blob is re-read 3-6 times and rewritten per hook.

Scaling: live rx 487 KB/s at about 2.5 hook events/s is about 195 KB of parameters per
event, consistent with whole-blob variable rewrites; tx 6 MB/s is consistent with
6-8 full row reads plus 3-6 blob reads per event on rows and blobs of tens of KB,

## Appendix A: Phase-1 sampler scripts

Save each block under its file name in one scratch directory (`$D` below). All four

`cpusample.py`:

```python
"""Read-only CPU attribution sampler: polls `ps` and sums CPU-time deltas per process.

Usage: python3 -I cpusample.py <seconds> <interval> <out.json>
"""

import json
import subprocess
import sys
import time


def parse_time(s: str) -> float:
    days = 0
    if "-" in s:
        d, s = s.split("-", 1)
        days = int(d)
    secs = 0.0
    for p in (float(p) for p in s.split(":")):
        secs = secs * 60 + p
    return days * 86400 + secs


def snap() -> dict[int, tuple[int, float, str]]:
    out = subprocess.run(
        ["ps", "-Ao", "pid=,ppid=,time=,command="], capture_output=True, text=True
    ).stdout
    rows = {}
    for line in out.splitlines():
        f = line.split(None, 3)
        if len(f) == 4:
            rows[int(f[0])] = (int(f[1]), parse_time(f[2]), f[3][:300])
    return rows


def main() -> None:
    total, interval, path = float(sys.argv[1]), float(sys.argv[2]), sys.argv[3]
    first: dict[int, float] = {}
    last: dict[int, tuple[int, float, str]] = {}
    seen_new: set[int] = set()
    start = time.time()
    for pid, row in snap().items():
        first[pid] = row[1]
        last[pid] = row
    samples = 0
    while time.time() - start < total:
        time.sleep(interval)
        samples += 1
        for pid, row in snap().items():
            if pid not in first:
                first[pid] = 0.0  # born during the window: all its CPU counts
                seen_new.add(pid)
            last[pid] = row
    elapsed = time.time() - start
    procs = [
        {"pid": pid, "ppid": ppid, "cpu_s": round(t - first[pid], 2), "new": pid in seen_new, "cmd": cmd}
        for pid, (ppid, t, cmd) in last.items()
        if t - first[pid] > 0
    ]
    procs.sort(key=lambda p: -p["cpu_s"])
    with open(path, "w") as fh:
        json.dump({"elapsed": elapsed, "samples": samples, "new_pids": len(seen_new), "procs": procs}, fh, indent=1)
    print(f"elapsed={elapsed:.0f}s samples={samples} new_pids={len(seen_new)}")


main()
```

`agg.py` groups a `cpusample.py` report by consumer:

```python
"""Usage: python3 -I agg.py <cpusample.json> <daemon_pid>"""

import collections
import json
import sys

with open(sys.argv[1]) as fh:
    d = json.load(fh)
el = d["elapsed"]
DAEMON = int(sys.argv[2])


def cat(p: dict) -> str:
    c = p["cmd"]
    exe = c.split()[0].rsplit("/", 1)[-1] if c.split() else "?"
    if "Virtualization.framework" in c:
        return "docker-vm"
    if "com.docker" in c or "Docker.app" in c:
        return "docker-other"
    if "/.gobby/bin/gterm" in c.split()[0]:
        return "gterm"
    if p["pid"] == DAEMON:
        return "daemon-main"
    if p["ppid"] == DAEMON and "codex" in c:
        return "daemon-textgen-codex"
    if p["ppid"] == DAEMON and "claude" in c:
        return "daemon-textgen-claude"
    if p["ppid"] == DAEMON:
        return "daemon-child:" + exe
    if "ghook" in c:
        return "ghook"
    if "pytest" in c:
        return "pytest"
    if exe in {"claude", "codex"}:
        return exe + "-cli"
    if "gcode" in exe:
        return "gcode"
    if exe in {"rustc", "cargo", "clippy-driver"}:
        return "rust-build"
    if exe in {"node", "npm"}:
        return "node"
    if exe.startswith("python"):
        return "python-other"
    if exe in {"tmux", "git"}:
        return exe
    return "other:" + exe[:30]


agg: dict[str, list[float]] = collections.defaultdict(lambda: [0.0, 0, 0])
for p in d["procs"]:
    a = agg[cat(p)]
    a[0] += p["cpu_s"]
    a[1] += 1
    a[2] += p["new"]
tot = sum(a[0] for a in agg.values())
print(f"elapsed={el:.0f}s total_cpu={tot:.0f}s avg_cores={tot / el:.2f} new_pids_seen={d['new_pids']}")
for k, (s, n, new) in sorted(agg.items(), key=lambda kv: -kv[1][0])[:32]:
    print(f"{s / el * 100:7.1f}%  {s:8.1f}s  procs={n:4d} new={new:4d}  {k}")
```

`spawnsample.py`:

```python
"""Read-only spawn-churn sampler: polls `ps` fast, counts newly seen pids by command and parent.

Usage: python3 -I spawnsample.py <seconds> <interval> <out.json>
"""

import collections
import json
import subprocess
import sys
import time


def snap() -> dict[int, tuple[int, str]]:
    out = subprocess.run(["ps", "-Ao", "pid=,ppid=,command="], capture_output=True, text=True).stdout
    rows = {}
    for line in out.splitlines():
        f = line.split(None, 2)
        if len(f) == 3:
            rows[int(f[0])] = (int(f[1]), f[2])
    return rows


def key(cmd: str) -> str:
    parts = cmd.split()
    exe = parts[0].rsplit("/", 1)[-1] if parts else "?"
    if exe.startswith("python") and len(parts) > 2:
        return f"{exe} {' '.join(parts[1:4])[:120]}"
    if exe in {"ghook", "tmux", "git", "gcode", "codex", "node", "uv"} and len(parts) > 1:
        return f"{exe} {' '.join(parts[1:3])[:80]}"
    return exe


def main() -> None:
    total, interval, path = float(sys.argv[1]), float(sys.argv[2]), sys.argv[3]
    base = snap()
    names = {p: c for p, (_, c) in base.items()}
    seen = set(base)
    by_cmd: collections.Counter[str] = collections.Counter()
    by_parent: collections.Counter[str] = collections.Counter()
    start = time.time()
    while time.time() - start < total:
        time.sleep(interval)
        for pid, (ppid, cmd) in snap().items():
            names.setdefault(pid, cmd)
            if pid in seen:
                continue
            seen.add(pid)
            k = key(cmd)
            by_cmd[k] += 1
            by_parent[f"{key(names.get(ppid, '?'))} -> {k.split()[0]}"] += 1
    elapsed = time.time() - start
    report = {
        "elapsed": elapsed,
        "seen_new": sum(by_cmd.values()),
        "by_cmd": by_cmd.most_common(60),
        "by_parent": by_parent.most_common(40),
    }
    with open(path, "w") as fh:
        json.dump(report, fh, indent=1)
    print(f"elapsed={elapsed:.0f}s seen_new={report['seen_new']}")


main()
```

`dstats.py`:

```python
"""Container CPU and network byte rates from the Docker Engine API (no DB access).

Usage: python3 -I dstats.py <seconds> <container>...
"""

import http.client
import json
import os
import socket
import sys
import time

SOCK = os.path.expanduser("~/.docker/run/docker.sock")


class UnixConn(http.client.HTTPConnection):
    def __init__(self, path: str) -> None:
        super().__init__("localhost")
        self.unix_path = path

    def connect(self) -> None:
        self.sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        self.sock.connect(self.unix_path)


def stats(name: str) -> dict:
    c = UnixConn(SOCK)
    c.request("GET", f"/containers/{name}/stats?stream=false&one-shot=true")
    d = json.loads(c.getresponse().read())
    net = d.get("networks", {})
    blk = d.get("blkio_stats", {}).get("io_service_bytes_recursive") or []
    return {
        "cpu_ns": d["cpu_stats"]["cpu_usage"]["total_usage"],
        "rx": sum(n["rx_bytes"] for n in net.values()),
        "tx": sum(n["tx_bytes"] for n in net.values()),
        "blk_w": sum(b["value"] for b in blk if b.get("op", "").lower() == "write"),
        "t": time.time(),
    }


secs = float(sys.argv[1])
names = sys.argv[2:]
a = {n: stats(n) for n in names}
time.sleep(secs)
b = {n: stats(n) for n in names}
for n in names:
    dt = b[n]["t"] - a[n]["t"]
    print(
        f"{n}: cpu={(b[n]['cpu_ns'] - a[n]['cpu_ns']) / 1e9 / dt * 100:.1f}% "
        f"rx={(b[n]['rx'] - a[n]['rx']) / dt / 1e3:.0f}KB/s "
        f"tx={(b[n]['tx'] - a[n]['tx']) / dt / 1e3:.0f}KB/s "
        f"blk_w={(b[n]['blk_w'] - a[n]['blk_w']) / dt / 1e3:.0f}KB/s over {dt:.0f}s"
    )
```
